{% folding, 导入马善政体： %}
```ts
//app/layout.tsx
import {Ma_Shan_Zheng} from "next/font/google";
```
{% endfolding %}
<hr>
{% folding, 定义字体： %}
```ts
const font = Ma_Shan_Zheng({
  weight: "400",
  subsets: ["latin"],
  variable: "--font-mashan",
  display: "block",
  preload: true
});
```
{% endfolding %}
<hr>
{% folding, 全局引入： %}
```ts
<html lang="zh-CN" className={font.className}>
   <body>{children}</body>
</html>
```
{% endfolding %}
