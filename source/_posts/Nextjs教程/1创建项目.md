{% folding, 关联远程仓库： %}
{% code %}
git init //进入根目录初始化git
git branch --show-current //查看本地当前分支
git branch -m master main //把当前master分支更改成main分支
git remote add origin git@github.com:wealipay/nextjs.git //与远程同名仓库关联
git remote -v //验证远程仓库是否添加成功
git add . 
git commit -m "项目初始化"
git pull origin main
git config pull.rebase false
git config pull.rebase true
git config pull.ff only
git pull origin main
git merge --no-ff
git branch --set-upstream-to=origin/main main
{% endcode %}
{% endfolding %}
