{% folding 项目安装: %}
```bash
npx create-next-app@latest
```
```bash
//禁止next.js收集数据
npx next telemetry disable
//查看收集状态
npx next telemetry status
```

{% endfolding %}

{% folding 生产依赖安装包: %}
```bash
pnpm install @prisma/client axios bcryptjs uuid nodemailer next-auth dotenv
```
{% endfolding %}
{% folding 开发依赖安装包: %}
```bash
pnpm install @types/nodemailer prisma @types/bcryptjs
```
{% endfolding %}

{% folding Prisma使用: %}
{% folding prisma初始化: %}
```
npx prisma init
```

初始化会生成
{% folding prisma/schema.prisma: %}
```ts
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

```
{% endfolding %}

{% folding .env: %}
```ts
DATABASE_URL=""

NODE_ENV="development"
```
{% endfolding %}


{% folding prisma.config.ts: %}
```ts
import "dotenv/config";
import { defineConfig } from "prisma/config";

export default defineConfig({
  schema: "prisma/schema.prisma",
  migrations: {
    path: "prisma/migrations",
  },
  datasource: {
    url: process.env["DATABASE_URL"],
  },
});

```
{% endfolding %}

{% endfolding %}

{% folding prisma文件格式化: %}
#### 安装:
```bash
pnpm install -D prettier prettier-plugin-prisma
```
<hr />

#### 配置：
```json
//.prettierrc
{
  "plugins": ["prettier-plugin-prisma"],
  "tabWidth": 2
}
```
<hr/>

#### 使用:
```bash
//在终端中输入
npx prettier --write **/*.prisma
```
